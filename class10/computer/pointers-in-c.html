1. How is a pointer variable different from a normal variable?

A normal variable stores a value, while a pointer variable stores the address of another variable.

int x = 10;
int *p = &x;
  • x → stores the value 10
  • p → stores the address of x
  • *p → gives the value stored at that address, i.e. 10
Variable → stores value
Pointer → stores address

2. Why is dynamic memory allocation an efficient memory management technique?

  1. Memory is allocated only when required.
  2. We can allocate memory according to the required size.
  3. Memory can be released when it is no longer needed.
  4. It avoids wasting a large amount of unused memory.
  5. It is useful when the required memory size is not known in advance.

Functions such as malloc(), calloc(), realloc() and free() are used for dynamic memory allocation.

3. How many bytes are needed to store an int pointer? Is it the same for a char pointer?

The size of a pointer depends mainly on the computer system, not on the data type it points to.

On many modern 64-bit systems:

  • int * → usually 8 bytes
  • char * → usually 8 bytes

Both store an address, so their sizes are usually the same.

#include <stdio.h>

int main()
{
    int *p1;
    char *p2;

    printf("Size of int pointer = %zu bytes\n", sizeof(p1));
    printf("Size of char pointer = %zu bytes\n", sizeof(p2));

    return 0;
}
Do not assume that a pointer is always 8 bytes. Its size depends on the system/compiler.

4. Write the output of the following code segments.

(a)

int *ptr, x = 9;

ptr = &x;

printf("\n %d", (*ptr)++);

Output: 9

(*ptr)++ first uses the value 9 and then increases it to 10. After the statement, x becomes 10.

(b)

int *ptr, x = 9;

ptr = &x;

printf("\n %d", (*ptr)++);

printf("\n %d", ptr);

The first printf() prints 9. The second is incorrect because ptr contains an address but %d is being used to print it. This gives undefined behaviour.

Correct way:

printf("\n %p", (void *)ptr);

(c)

int *ptr, x = 9;

ptr = &x;

int y = ++(*ptr);

printf("\n %d", y);

Output: 10

++(*ptr) is pre-increment: 9 → 10, then 10 is assigned to y.

(d)

char *ptr, x = 'A';

ptr = &x;

char y = *ptr;

printf("\n %c", y);

Output: A

(e)

char *ptr, x = 'A';

ptr = &x;

char y = (*ptr)++;

printf("\n %c", y);

Output: A

(*ptr)++ is post-increment. It gives A first, then changes x to B.

(f)

char *ptr, x = 'A';

ptr = &x;

char y = ++(*ptr);

printf("\n %c", y);

Output: B

++(*ptr) first changes A to B and then assigns it to y.

(g)

char *ptr, x = 'A';

ptr = &x;

char *y;

y = ptr;

printf("\n %c", ++(*y));

Output: B

y = ptr means both pointers point to x. Then ++(*y) changes A → B.

5. Dynamically allocate memory for an array of 10 integers and display the first 5.

#include <stdio.h>
#include <stdlib.h>

int main()
{
    int *a;
    int i;

    a = (int *)malloc(10 * sizeof(int));

    if(a == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    printf("Enter 10 integers:\n");

    for(i = 0; i < 10; i++)
        scanf("%d", &a[i]);

    printf("First 5 elements:\n");

    for(i = 0; i < 5; i++)
        printf("%d ", a[i]);

    free(a);

    return 0;
}

malloc(10 * sizeof(int)) allocates memory for 10 integers. free(a) releases the allocated memory.

6. Dynamically allocate an array to store runs scored in the last ten ODI matches and find the maximum.

#include <stdio.h>
#include <stdlib.h>

int maximum(int a[], int n)
{
    int i;
    int max = a[0];

    for(i = 1; i < n; i++)
    {
        if(a[i] > max)
            max = a[i];
    }

    return max;
}

int main()
{
    int *runs;
    int i, max;

    runs = (int *)malloc(10 * sizeof(int));

    if(runs == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    printf("Enter runs in 10 matches:\n");

    for(i = 0; i < 10; i++)
        scanf("%d", &runs[i]);

    max = maximum(runs, 10);

    printf("Maximum runs = %d", max);

    free(runs);

    return 0;
}
  • Store 10 scores dynamically.
  • Compare every score with max.
  • Keep the largest value.
  • Return the maximum score.

7. Allocate memory dynamically to store your name and write another function to display it.

#include <stdio.h>
#include <stdlib.h>

char *createName(int length)
{
    char *name;

    name = (char *)malloc((length + 1) * sizeof(char));

    return name;
}

void displayName(char *name)
{
    printf("Name = %s", name);
}

int main()
{
    int length;
    char *name;

    printf("Enter length of your name: ");
    scanf("%d", &length);

    name = createName(length);

    if(name == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    printf("Enter your name: ");
    scanf("%s", name);

    displayName(name);

    free(name);

    return 0;
}

Why length + 1? A C string needs one extra space for the null character '\0' at the end.

8. Store integers in an array, count the even numbers, dynamically allocate a second array for only the even numbers, and copy them.

First array: 2 4 0 9 1 9 8 6

Second array: 2 4 0 8 6

#include <stdio.h>
#include <stdlib.h>

int countEven(int a[], int n)
{
    int i, count = 0;

    for(i = 0; i < n; i++)
    {
        if(a[i] % 2 == 0)
            count++;
    }

    return count;
}

void copyEven(int a[], int b[], int n)
{
    int i, j = 0;

    for(i = 0; i < n; i++)
    {
        if(a[i] % 2 == 0)
        {
            b[j] = a[i];
            j++;
        }
    }
}

int main()
{
    int a[] = {2, 4, 0, 9, 1, 9, 8, 6};
    int n = 8;
    int count, *b;
    int i;

    count = countEven(a, n);
    b = (int *)malloc(count * sizeof(int));

    if(b == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    copyEven(a, b, n);

    printf("First array:\n");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n\nSecond array:\n");
    for(i = 0; i < count; i++)
        printf("%d ", b[i]);

    free(b);

    return 0;
}

Output:

First array:
2 4 0 9 1 9 8 6

Second array:
2 4 0 8 6

9. Store integers, select non-zero elements divisible by 3, dynamically allocate a second array, copy them, and calculate their sum.

First array: 2 4 0 9 1 9 8 6

Selected elements: 9 9 6

Sum: 24

#include <stdio.h>
#include <stdlib.h>

int countDivisibleBy3(int a[], int n)
{
    int i, count = 0;

    for(i = 0; i < n; i++)
    {
        if(a[i] != 0 && a[i] % 3 == 0)
            count++;
    }

    return count;
}

void copyElements(int a[], int b[], int n)
{
    int i, j = 0;

    for(i = 0; i < n; i++)
    {
        if(a[i] != 0 && a[i] % 3 == 0)
        {
            b[j] = a[i];
            j++;
        }
    }
}

int sumElements(int b[], int n)
{
    int i, sum = 0;

    for(i = 0; i < n; i++)
        sum = sum + b[i];

    return sum;
}

int main()
{
    int a[] = {2, 4, 0, 9, 1, 9, 8, 6};
    int n = 8;
    int count, *b;
    int sum, i;

    count = countDivisibleBy3(a, n);
    b = (int *)malloc(count * sizeof(int));

    if(b == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    copyElements(a, b, n);
    sum = sumElements(b, count);

    printf("First array:\n");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\n\nSecond array:\n");
    for(i = 0; i < count; i++)
        printf("%d ", b[i]);

    printf("\n\nSummation = %d", sum);

    free(b);

    return 0;
}

Output:

First array:
2 4 0 9 1 9 8 6

Second array:
9 9 6

Summation = 24

⭐ Quick Revision — Pointers

1. Pointer

A pointer is a variable that stores the address of another variable.

int x = 10;
int *p = &x;

2. & operator

& gives the address of a variable.

&p

3. * operator

* is used to access the value stored at an address.

*p

4. Example

int x = 10;
int *p = &x;

x → 10
&p → address of p
p → address of x
*p → 10

5. Dynamic memory functions

FunctionUse
malloc()Allocates memory
calloc()Allocates and initializes memory
realloc()Changes allocated memory size
free()Releases allocated memory
int *p;

p = (int *)malloc(10 * sizeof(int));

free(p);

malloc() → get memory
free() → release memory
