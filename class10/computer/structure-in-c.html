1. Define structure in the context of a programming language. How is it different from an array?

A structure is a user-defined data type in C that allows us to store different types of data under one name.

struct Student
{
    char name[30];
    int age;
    float marks;
};
StructureArray
Can store different data types.Normally stores elements of the same data type.
Members can be int, float, char, etc.All elements have the same data type.
Members are accessed using .Elements are accessed using an index [ ].
Example: student name, age and marks.Example: marks of many students.
Array → Same type of data
Structure → Different types of data

2. Is structure a built-in data type? Can we apply arithmetic operations to structure variables?

  • No, structure is not a built-in data type.
  • It is a user-defined data type.
  • We cannot directly perform addition or subtraction on two complete structure variables.
  • We can perform arithmetic operations on their individual members.
#include <stdio.h>

struct Number
{
    int a;
    int b;
};

int main()
{
    struct Number n1, n2, result;

    n1.a = 10;
    n1.b = 20;

    n2.a = 5;
    n2.b = 10;

    result.a = n1.a + n2.a;
    result.b = n1.b + n2.b;

    printf("a = %d\n", result.a);
    printf("b = %d\n", result.b);

    return 0;
}

Output:

a = 15
b = 30

This is not allowed:

result = n1 + n2;

4. Identify the structure variables.

struct Account account1, account2, account3, account[10];

The structure variables are:

  1. account1
  2. account2

account1, account2, and account3 are individual structure variables. account[10] is an array of 10 structure variables.

6. Declare a pointer for Teacher and dynamically allocate memory for 10 records.

teacher = (struct Teacher *)malloc(10 * sizeof(struct Teacher));

Include:

#include <stdlib.h>

Complete small example:

#include <stdio.h>
#include <stdlib.h>

struct Teacher
{
    char name[30];
    int age;
};

int main()
{
    struct Teacher *teacher;

    teacher = (struct Teacher *)malloc(
        10 * sizeof(struct Teacher)
    );

    if(teacher == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    free(teacher);

    return 0;
}

7. Consider struct Employee and write statements/programs.

struct Employee
{
    char name[30];
    double salary;
};

(a) Declare a pointer and display the salary.

struct Employee emp;
struct Employee *ptr;

ptr = &emp;

printf("%lf", ptr->salary);

When using a structure pointer, we use the arrow -> operator. ptr->salary means the salary member of the structure pointed to by ptr.

(b) Use a single pointer for two structure variables and display the employee whose salary is more.

#include <stdio.h>

struct Employee
{
    char name[30];
    double salary;
};

int main()
{
    struct Employee e1 = {"Rahul", 40000};
    struct Employee e2 = {"Amit", 50000};

    struct Employee *ptr;

    if(e1.salary > e2.salary)
        ptr = &e1;
    else
        ptr = &e2;

    printf("Name: %s\n", ptr->name);
    printf("Salary: %.2lf\n", ptr->salary);

    return 0;
}

Output:

Name: Amit
Salary: 50000.00

8. Rewrite Q. 7 for dynamic memory allocation for N records where N is user input.

#include <stdio.h>
#include <stdlib.h>

struct Employee
{
    char name[30];
    double salary;
};

int main()
{
    struct Employee *emp;
    int n, i;

    printf("Enter number of employees: ");
    scanf("%d", &n);

    emp = (struct Employee *)malloc(
        n * sizeof(struct Employee)
    );

    if(emp == NULL)
    {
        printf("Memory allocation failed");
        return 1;
    }

    for(i = 0; i < n; i++)
    {
        printf("\nEnter name: ");
        scanf("%s", emp[i].name);

        printf("Enter salary: ");
        scanf("%lf", &emp[i].salary);
    }

    printf("\nEmployee Details:\n");

    for(i = 0; i < n; i++)
    {
        printf("\nName: %s", emp[i].name);
        printf("\nSalary: %.2lf\n", emp[i].salary);
    }

    free(emp);

    return 0;
}
malloc(n * sizeof(struct Employee)) allocates memory for N employee records.

9. Design a simple banking system using structures.

The program supports:

  1. Opening an account
  2. Displaying details using account number
  3. Displaying all account details
  4. Displaying accounts with balance more than 1000
  5. Depositing money
  6. Withdrawing money
#include <stdio.h>
#include <string.h>

#define MAX 100

struct Account
{
    char acNo[15];
    char ifsc[15];
    char acType[7];
    double balance;
    double minBalance;
};

struct Account accounts[MAX];
int count = 0;

void openAccount()
{
    if(count >= MAX)
    {
        printf("Account limit reached.\n");
        return;
    }

    printf("\nEnter Account Number: ");
    scanf("%14s", accounts[count].acNo);

    printf("Enter IFSC Code: ");
    scanf("%14s", accounts[count].ifsc);

    printf("Enter Account Type: ");
    scanf("%6s", accounts[count].acType);

    printf("Enter Minimum Balance: ");
    scanf("%lf", &accounts[count].minBalance);

    printf("Enter Initial Balance: ");
    scanf("%lf", &accounts[count].balance);

    count++;
    printf("Account opened successfully.\n");
}

int findAccount(char acNo[])
{
    int i;

    for(i = 0; i < count; i++)
    {
        if(strcmp(accounts[i].acNo, acNo) == 0)
            return i;
    }

    return -1;
}

void displayAccount(int i)
{
    printf("\nAccount Number : %s", accounts[i].acNo);
    printf("\nIFSC Code     : %s", accounts[i].ifsc);
    printf("\nAccount Type   : %s", accounts[i].acType);
    printf("\nBalance        : %.2lf", accounts[i].balance);
    printf("\nMinimum Balance: %.2lf\n", accounts[i].minBalance);
}

void displayByAccountNumber()
{
    char acNo[15];
    int index;

    printf("\nEnter Account Number: ");
    scanf("%14s", acNo);

    index = findAccount(acNo);

    if(index == -1)
        printf("Account not found.\n");
    else
        displayAccount(index);
}

void displayAll()
{
    int i;

    if(count == 0)
    {
        printf("No accounts available.\n");
        return;
    }

    for(i = 0; i < count; i++)
        displayAccount(i);
}

void displayAbove1000()
{
    int i;

    for(i = 0; i < count; i++)
    {
        if(accounts[i].balance > 1000)
            displayAccount(i);
    }
}

void deposit()
{
    char acNo[15];
    double amount;
    int index;

    printf("\nEnter Account Number: ");
    scanf("%14s", acNo);

    index = findAccount(acNo);

    if(index == -1)
    {
        printf("Account not found.\n");
        return;
    }

    printf("Enter amount to deposit: ");
    scanf("%lf", &amount);

    if(amount <= 0)
    {
        printf("Invalid amount.\n");
        return;
    }

    accounts[index].balance += amount;

    printf("Deposit successful.\n");
    printf("New Balance = %.2lf\n", accounts[index].balance);
}

void withdraw()
{
    char acNo[15];
    double amount;
    int index;

    printf("\nEnter Account Number: ");
    scanf("%14s", acNo);

    index = findAccount(acNo);

    if(index == -1)
    {
        printf("Account not found.\n");
        return;
    }

    printf("Enter amount to withdraw: ");
    scanf("%lf", &amount);

    if(amount <= 0)
    {
        printf("Invalid amount.\n");
        return;
    }

    if(accounts[index].balance - amount < accounts[index].minBalance)
    {
        printf("Withdrawal not allowed.\n");
        printf("Minimum balance must be maintained.\n");
        return;
    }

    accounts[index].balance -= amount;

    printf("Withdrawal successful.\n");
    printf("New Balance = %.2lf\n", accounts[index].balance);
}

int main()
{
    int choice;

    do
    {
        printf("\n===== BANKING SYSTEM =====\n");
        printf("1. Open Account\n");
        printf("2. Display Account\n");
        printf("3. Display All Accounts\n");
        printf("4. Display Accounts with Balance > 1000\n");
        printf("5. Deposit\n");
        printf("6. Withdraw\n");
        printf("7. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1: openAccount(); break;
            case 2: displayByAccountNumber(); break;
            case 3: displayAll(); break;
            case 4: displayAbove1000(); break;
            case 5: deposit(); break;
            case 6: withdraw(); break;
            case 7: printf("Thank you!\n"); break;
            default: printf("Invalid choice.\n");
        }
    } while(choice != 7);

    return 0;
}

How this program works

  1. Open account: stores account number, IFSC, account type, minimum balance and initial balance.
  2. Display by account number: searches for the account number and displays its details.
  3. Display all: displays every stored account.
  4. Balance greater than 1000: checks accounts[i].balance > 1000.
  5. Deposit: adds the amount to the balance.
  6. Withdrawal: checks that the minimum balance will be maintained before subtracting the amount.

Use the arrow (->) operator:

ptr->age

struct Student *s;

Free memory

free(s);

Array → same type of data
. → normal structure variable
-> → structure pointer
